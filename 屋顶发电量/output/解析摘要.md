### panelLocationInfos 结构
```json
{
  "positions": [x1, y1, z1, x2, y2, z2, ...],
  "aspect": 237.6,          // 朝向角度
  "slope": 0.401,           // 倾斜角度
  "start": [x, y, z],       // 起始点
  "end": [x, y, z],         // 结束点
  "center": [x, y, z],      // 中心点
  "generationPowerVO": {    // 发电量数据（仅在 layout 中）
    "calStatus": true,
    "monthlyHourlyPowerList": [[...], ...],  // 12月 × 24小时
    "monthlyPowerList": [...],               // 12个月
    "yearPower": 14.5,                       // 年度总发电量
    "dailyPowerList": [...]                  // 365天
  }
}
```
